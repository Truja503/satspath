\hypertarget{satspath-demo-flow}{%
\section{SatsPath Demo Flow}\label{satspath-demo-flow}}

This document shows a complete example session using the SatsPath CLI.

\hypertarget{setup}{%
\subsection{Setup}\label{setup}}

\begin{Shaded}
\begin{Highlighting}[]
\BuiltInTok{cd}\NormalTok{ \textasciitilde{}/Desktop/satspath}
\ExtensionTok{cargo}\NormalTok{ build }\AttributeTok{{-}{-}release}
\end{Highlighting}
\end{Shaded}

\hypertarget{step-by-step}{%
\subsection{Step-by-step}\label{step-by-step}}

\hypertarget{initialize}{%
\subsubsection{1. Initialize}\label{initialize}}

\begin{Shaded}
\begin{Highlighting}[]
\ExtensionTok{satspath}\NormalTok{ init}
\end{Highlighting}
\end{Shaded}

Output:

\begin{verbatim}
Created .satspath/registry.json
Created .satspath/keys.json

SatsPath initialized at .satspath/
NOTE: .satspath/ is git-ignored and stays local to this machine.
\end{verbatim}

\hypertarget{register}{%
\subsubsection{2. Register}\label{register}}

\begin{Shaded}
\begin{Highlighting}[]
\ExtensionTok{satspath}\NormalTok{ register rodrigo@satspath.dev}
\end{Highlighting}
\end{Shaded}

Output:

\begin{verbatim}
Registered: rodrigo@satspath.dev
Identity pubkey: 02a1b2c3...
Fingerprint:     a1b2c3d4

Payment methods registered:
  - Lightning Address
  - Bitcoin on-chain (primary)
  - Bitcoin on-chain (secondary, privacy address)
  - Ark

Profile signed and stored in .satspath/registry.json
DEMO secret key stored in .satspath/keys.json (git-ignored)
\end{verbatim}

\subsubsection{3. Show}\label{show}}

      Address: bc1qa1b2...
  - Bitcoin (secondary) [On-chain]
      Address: bc1qb3c4...
  - Ark [Ark]
      Server: ark.satspath.dev
\end{verbatim}

\subsubsection{4. Encode a payment

\begin{Shaded}

\end{Shaded}

Output:

\begin{verbatim}
Encoded SatsPath URI:

\end{verbatim}

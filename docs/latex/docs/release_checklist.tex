\hypertarget{security-and-privacy}{%
\subsection{Security and Privacy}\label{security-and-privacy}}

\begin{itemize}
\tightlist
\item[$\boxtimes$]
  All cryptographic implementations reviewed.
\item[$\boxtimes$]
  No private material (seeds, keys) included in profiles or payment
  pointers.
\item[$\boxtimes$]
  Sensitive user data masked in preview modes.
\end{itemize}

\hypertarget{functional-testing}{%
\subsection{Functional Testing}\label{functional-testing}}

\begin{itemize}
\tightlist
\item[$\boxtimes$]
  All unit tests pass across workspaces.
\item[$\boxtimes$]
  Integration tests for HTTP resolver and LNURL pass.
\item[$\boxtimes$]
  Testnet execution paths enabled; mainnet execution paths strictly
  disabled.
\end{itemize}

\hypertarget{code-quality}{%
\subsection{Code Quality}\label{code-quality}}

\begin{itemize}
\tightlist

  CI workflows in place for formatting, linting, and testing.
\item[$\boxtimes$]
  PR template and branch protection guidelines established.
\end{itemize}

\hypertarget{documentation}{%
\subsection{Documentation}\label{documentation}}

\begin{itemize}
\tightlist

\end{itemize}
